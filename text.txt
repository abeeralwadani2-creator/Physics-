  <script>
    // إنشاء نجوم العلفية عشوائياً
    const spaceBg = id('spaceBg');
    for (let i = 0; i < 70; i++) {
      const star = document.createElement('div');
      star.className = 'star';
      star.style.width = Math.random() * 3 + 'px';
      star.style.height = star.style.width;
      star.style.left = Math.random() * 100 + '%';
      star.style.top = Math.random() * 100 + '%';
      star.style.setProperty('--duration', (Math.random() * 3 + 2) + 's');
      spaceBg.appendChild(star);
    }

    // بيانات المراحل والأسئلة
    const stages = [
      {
        question: "المرحلة الأولى: ما القوة التي تساعد الصاروخ على الانطلاق إلى أعلى؟",
        options: [
          { text: "🌍 الجاذبية", correct: false },
          { text: "⬆️ قوة الدفع", correct: true },
          { text: "💨 الاحتكاك", correct: false },

        ],
        successMsg: "✅ صحيح! قوة الدفع تساعد الصاروخ على الانطلاق."
      },
      {
        question: "المرحلة الثانية: تندفع الغازات إلى أسفل، فيتحرك الصاروخ إلى أعلى. أي قانون يفسر ذلك؟",
        options: [

        ],

      },
      {
        question: "المرحلة الثالثة: ما القوة التي تسحب الصاروخ باتجاه الأرض أثناء صعوده؟",
        options: [
          { text: "قوة الدفع", correct: false },
          { text: "الجاذبية", correct: true },
          { text: "القوة الكهربائية", correct: false },
          { text: "المغناطيسية", correct: false }
        ],
        successMsg: "✅ صحيح! الجاذبية تجذب الصاروخ نحو الأرض."
      },
      {
        question: "المرحلة الرابعة: لكي يدخل الصاروخ في مدار حول الأرض، ما الذي يجب أن يكون مناسبًا؟",
        options: [
          { text: "السرعة والاتجاه", correct: true },
          { text: "لون الصاروخ", correct: false },
          { text: "درجة الحرارة فقط", correct: false },
          { text: "شكل الصاروخ", correct: false }
        ],
        successMsg: "✅ صحيح! السرعة والاتجاه المناسبان يساعدان الصاروخ على الوصول إلى المدار."
      }
    ];

    let currentStage = 0;
    let isAnswering = false;

    function id(elemId) { return document.getElementById(elemId); }

    function showScreen(screenId) {
      document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
      id(screenId).classList.add('active');
    }

    function startGame() {
      currentStage = 0;
      showScreen('game-screen');
      updateRocketPosition();
      loadStage();
    }

    function loadStage() {
      isAnswering = false;
      id('feedbackText').innerText = '';
      id('feedbackText').className = 'feedback';
      
      const stageData = stages[currentStage];
      id('questionText').innerText = stageData.question;
      
      const optionsBox = id('optionsBox');
      optionsBox.innerHTML = '';

      stageData.options.forEach(opt => {
        const btn = document.createElement('button');
        btn.className = 'btn';
        btn.innerText = opt.text;
        btn.onclick = () => handleAnswer(opt, btn);
        optionsBox.appendChild(btn);
      });
    }

    function handleAnswer(option, button) {
      if (isAnswering) return;

      if (option.correct) {
        isAnswering = true;
        button.classList.add('correct');
        id('feedbackText').innerText = stages[currentStage].successMsg;
        id('feedbackText').className = 'feedback correct-text';

        // الانتقال للمرحلة التالية
        currentStage++;
        updateRocketPosition();

        setTimeout(() => {
          if (currentStage < stages.length) {
            loadStage();
          } else {
            showScreen('end-screen');
          }
        }, 1800);

      } else {
        button.classList.add('wrong');
        id('feedbackText').innerText = "❌ حاول مرة أخرى!";
        id('feedbackText').className = 'feedback wrong-text';
        
        setTimeout(() => {
          button.classList.remove('wrong');
        }, 800);
      }
    }

    function updateRocketPosition() {
      // حساب النسبة المئوية لتحريك الصاروخ على الشريط (RTL)
      const percentages = [10, 30, 50, 70, 90]; 
      const progressWidths = [0, 25, 50, 75, 100];

      const rocket = id('rocketRunner');
      rocket.style.right = percentages[currentStage] + '%';

      id('trackProgress').style.width = progressWidths[currentStage] + '%';

      // تحديث حالة النقاط على المسار
      for (let i = 0; i <= 4; i++) {
        const node = id(`node-${i}`);
        if (i <= currentStage) {
          node.classList.add('reached');
        } else {
          node.classList.remove('reached');
        }
      }
    }

    function resetGame() {
      startGame();
    }
  </script>
